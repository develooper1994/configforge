# Register one Doctor invocation as a test.
#
# ~~~
# doctor_case(<name> <expected-exit-code> <arg>...)
# ~~~
function(doctor_case name expect_fail)
  add_test(
    NAME doctor.${name}
    COMMAND
      ${CMAKE_COMMAND} -DDOCTOR=$<TARGET_FILE:doctor> -DARGS=${ARGN}
      -DEXPECT_FAIL=${expect_fail} -P
      ${CMAKE_SOURCE_DIR}/tests/doctor/run.cmake)
  set_tests_properties(doctor.${name} PROPERTIES FIXTURES_REQUIRED
                                                 DOCTOR_REPORTS)
endfunction()

# A clean report exits 0, and one with a finding at the --fail-on level 1.
# The exit code is the interface a CI job uses, so it is what gets
# asserted.
doctor_case(lint_clean 0 lint|cf_fx_staging.json)
# A dead source is a warning by default: a system file wholly overridden
# by a user's is often intended. It fails a run that asks for warnings.
doctor_case(lint_dead 0 lint|cf_fx_dead.json)
doctor_case(lint_dead_strict 1 lint|cf_fx_dead.json|--fail-on|warning)
doctor_case(dump 0 dump|cf_fx_prod.json)
doctor_case(explain 0 explain|cf_fx_prod.json|server.port)
# A document says what it was read as and how it was chosen (D-73); a
# report from a configuration with none says nothing of the kind.
doctor_case(explain_document 0 explain|cf_fx_doc.json|server.port)
doctor_case(explain_missing 2 explain|cf_fx_prod.json|no.such.key)
doctor_case(diff 0 diff|cf_fx_prod.json|cf_fx_staging.json)
doctor_case(lint_policy 1 lint|cf_fx_leak.json)
doctor_case(lint_rules 0 lint|cf_fx_rules.json)
doctor_case(lint_twice 1 lint|cf_fx_twice.json)
doctor_case(lint_sources 1 lint|cf_fx_sources.json)
# A key nothing won is still a key: the secret it refused is found, and
# explain shows the candidate that was refused.
doctor_case(lint_unset 1 lint|cf_fx_unset.json)
doctor_case(explain_unset 0 explain|cf_fx_unset.json|license.code)
doctor_case(lint_files_secret 0 lint|cf_fx_files.json|--fail-on|never)
doctor_case(not_a_report 2 dump|cf_fx_staging.ini)
# A sequence, oldest first: what changed from each report to the next,
# and since which report a key has been what it is. A change is info, so
# it fails only a run that asks for info; a report that cannot be read,
# or a key none of them has, is unusable input.
doctor_case(history 0 history|cf_fx_prod.json|cf_fx_staging.json)
doctor_case(history_strict 1
            history|cf_fx_prod.json|cf_fx_staging.json|--fail-on|info)
doctor_case(history_unreadable 2 history|cf_fx_prod.json|cf_fx_none.json)
doctor_case(blame 0 blame|server.port|cf_fx_prod.json|cf_fx_staging.json)
doctor_case(blame_missing 2 blame|no.such.key|cf_fx_prod.json)
# Only what a change touched: nothing, against itself, so the info that a
# plain lint shows fails no run; a source the change went through keeps
# its warning; and the flag without a base is unusable.
set(only --only-changed)
doctor_case(
  only_changed_none 0
  lint|cf_fx_staging.json|--base|cf_fx_staging.json|${only}|--fail-on|info)
doctor_case(
  only_changed_source 1
  lint|cf_fx_dead.json|--base|cf_fx_prod.json|${only}|--fail-on|warning)
doctor_case(only_changed_alone 2 lint|cf_fx_staging.json|--only-changed)
doctor_case(base_alone 2 lint|cf_fx_staging.json|--base|cf_fx_prod.json)

# Files against a JSON Schema contract (contract.c): a file that holds to

            check|--schema|cf_fx_none.json|cf_fx_contract_ok.json)
doctor_case(check_alone 2 check|cf_fx_contract_ok.json)

# The contract as C (`doctor gen --schema`), generated by this build's
# Doctor and compiled with the project's warnings, against the schema it
# came from. Where Doctor runs on the build machine.
if(NOT CMAKE_CROSSCOMPILING)
  set(gen_c ${CMAKE_CURRENT_BINARY_DIR}/gen_contract.c)
  add_custom_command(
    OUTPUT ${gen_c}
    COMMAND doctor gen --schema ${contract} --name gen_contract --out ${gen_c}
    DEPENDS doctor ${contract}
    VERBATIM)
  add_executable(doctor_gen test_gen.c ${gen_c})
  target_compile_definitions(doctor_gen
                             PRIVATE CF_CONTRACT_SCHEMA="${contract}")
  target_link_libraries(doctor_gen PRIVATE configforge)
  set_target_properties(
    doctor_gen
    PROPERTIES C_STANDARD 90
               C_STANDARD_REQUIRED ON
               C_EXTENSIONS OFF)
  configforge_apply_warnings(doctor_gen)
  configforge_apply_analysis(doctor_gen)
  add_test(NAME doctor.gen COMMAND doctor_gen)
endif()

# The GitHub Action's scripts and the pre-commit hook (actions/doctor),
# as a project's CI runs them. Bash on a Unix host only: on Windows the
# bash on PATH may be WSL's, which would not see this build's paths.

  find_program(CF_BASH bash)
  if(CF_BASH)
    add_test(
      NAME doctor.action
      COMMAND
        ${CMAKE_COMMAND} -DDOCTOR=$<TARGET_FILE:doctor> -DBASH=${CF_BASH}
        -DACTION=${CMAKE_SOURCE_DIR}/actions/doctor
        -DWORK=${CMAKE_CURRENT_BINARY_DIR} -P
        ${CMAKE_CURRENT_SOURCE_DIR}/action.cmake)
    set_tests_properties(doctor.action PROPERTIES FIXTURES_REQUIRED
                                                  DOCTOR_REPORTS)
  endif()
endif()
